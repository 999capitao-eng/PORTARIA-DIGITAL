import logging
import sys

from PySide6.QtCore import QLibraryInfo, QLocale, QTranslator
from PySide6.QtWidgets import QApplication

from portaria.db import conectar, preparar_banco
from portaria.ui.janela_principal import JanelaPrincipal
from portaria.ui.tema import gerar_folha, tema_salvo

def main():
    logging.basicConfig(
        filename="portaria.log",
        level=logging.INFO,
        format="%(asctime)s %(levelname)s %(name)s: %(message)s",
    )
    conn = conectar()
    preparar_banco(conn)

    app = QApplication(sys.argv)

    # Botões padrão do Qt (Salvar, Cancelar, Sim, Não) em português
    tradutor = QTranslator(app)
    pasta_traducoes = QLibraryInfo.path(QLibraryInfo.LibraryPath.TranslationsPath)
    if tradutor.load(QLocale(QLocale.Language.Portuguese, QLocale.Country.Brazil),
                     "qtbase", "_", pasta_traducoes):
        app.installTranslator(tradutor)

    app.setStyle("Fusion")  # mesma aparência em Windows, Linux e macOS
    app.setStyleSheet(gerar_folha(tema_salvo()))  # tema escolhido na última execução

    janela = JanelaPrincipal(conn)
    janela.show()
    codigo = app.exec()
    conn.close()
    return codigo

if __name__ == "__main__":
    sys.exit(main())

def listar_historico(conn, inicio: str, fim: str, termo: str = "") -> list[Linha]:
    """Visitas com entrada no intervalo [inicio, fim], filtrando por nome ou setor."""
    padrao = f"%{termo.strip()}%"  # o curinga vai no PARÂMETRO, não no SQL
    return conn.execute(
        """
        SELECT v.id, p.nome, p.documento, s.nome AS setor, c.codigo AS cracha,
               v.entrada, v.saida,
               CAST(ROUND((julianday(v.saida) - julianday(v.entrada)) * 1440) AS INTEGER)
                   AS minutos
          FROM visita v
          JOIN visitante p ON p.id = v.visitante_id
          JOIN setor s     ON s.id = v.setor_id
          JOIN cracha c    ON c.id = v.cracha_id
         WHERE date(v.entrada) BETWEEN ? AND ?
           AND (p.nome LIKE ? OR s.nome LIKE ?)
         ORDER BY v.entrada DESC
        """,
        (inicio, fim, padrao, padrao),
    ).fetchall()

def resumo_por_setor(conn, inicio: str, fim: str) -> list[Linha]:
    return conn.execute(
        """
        SELECT s.nome AS setor, COUNT(v.id) AS visitas,
               ROUND(AVG((julianday(v.saida) - julianday(v.entrada)) * 1440), 1)
                   AS media_minutos
          FROM setor s
          LEFT JOIN visita v
                 ON v.setor_id = s.id AND date(v.entrada) BETWEEN ? AND ?
         GROUP BY s.id
         ORDER BY visitas DESC, s.nome
        """,
        (inicio, fim),
    ).fetchall()

def listar_setores(conn) -> list[Linha]:
    """Todos os setores, inclusive os inativos, para a tela de gestão."""
    return conn.execute(
        """
        SELECT s.id, s.nome, s.lotacao_maxima, s.ativo,
               COUNT(v.id) AS total_visitas
          FROM setor s
          LEFT JOIN visita v ON v.setor_id = s.id
         GROUP BY s.id
         ORDER BY s.nome
        """
    ).fetchall()

def inserir_setor(conn, nome: str, lotacao_maxima: int) -> int:
    cur = conn.execute(
        "INSERT INTO setor (nome, lotacao_maxima) VALUES (?, ?)", (nome, lotacao_maxima)
    )
    return cur.lastrowid

def atualizar_setor(conn, setor_id: int, nome: str, lotacao_maxima: int, ativo: bool) -> None:
    conn.execute(
        "UPDATE setor SET nome = ?, lotacao_maxima = ?, ativo = ? WHERE id = ?",
        (nome, lotacao_maxima, int(ativo), setor_id),
    )

def excluir_setor(conn, setor_id: int) -> None:
    conn.execute("DELETE FROM setor WHERE id = ?", (setor_id,))

def contar_visitas_do_setor(conn, setor_id: int) -> int:
    return conn.execute(
        "SELECT COUNT(*) FROM visita WHERE setor_id = ?", (setor_id,)
    ).fetchone()[0]

from portaria import repositorio as repo
from portaria.erros import RegraNegocioError
from portaria.validacao import normalizar_documento, normalizar_nome


# ---------- visitantes ----------

def cadastrar_visitante(conn, nome: str, documento: str) -> int:
    """Cadastra um visitante após validar os dados."""
    nome = normalizar_nome(nome)
    documento = normalizar_documento(documento)

    if not nome:
        raise RegraNegocioError("Informe o nome do visitante.")

    visitante = repo.buscar_visitante_por_documento(conn, documento)
    if visitante is not None:
        return visitante["id"]

    with conn:
        return repo.inserir_visitante(conn, nome, documento)


# ---------- visitas ----------

def registrar_entrada(conn, documento: str, setor_id: int) -> int:
    """Registra a entrada e associa um crachá disponível."""
    documento = normalizar_documento(documento)

    visitante = repo.buscar_visitante_por_documento(conn, documento)
    if visitante is None:
        raise RegraNegocioError(
            "Visitante não cadastrado. Informe o nome e cadastre-o primeiro."
        )

    visitante_id = visitante["id"]

    aberta = repo.buscar_visita_aberta_por_visitante(conn, visitante_id)
    if aberta is not None:
        raise RegraNegocioError(
            f"{visitante['nome']} já está no prédio. Registre a saída antes."
        )

    setor = next(
        (s for s in repo.listar_setores_ativos(conn) if s["id"] == setor_id),
        None,
    )
    if setor is None:
        raise RegraNegocioError("O setor selecionado não existe ou está inativo.")

    ocupacao = repo.contar_presentes_no_setor(conn, setor_id)
    if ocupacao >= setor["lotacao_maxima"]:
        raise RegraNegocioError(
            f"O setor {setor['nome']} atingiu sua lotação máxima."
        )

    cracha = repo.buscar_cracha_disponivel(conn)
    if cracha is None:
        raise RegraNegocioError(
            "Não há crachá disponível. Aguarde uma devolução."
        )

    with conn:
        visita_id = repo.inserir_visita(
            conn,
            visitante_id,
            setor_id,
            cracha["id"],
        )
        repo.atualizar_situacao_cracha(conn, cracha["id"], "em_uso")

    return visita_id


def registrar_saida(conn, visita_id: int) -> None:
    """Encerra uma visita e devolve o crachá na mesma transação."""
    visita = repo.buscar_visita_aberta_por_id(conn, visita_id)

    if visita is None:
        raise RegraNegocioError("Esta visita já foi encerrada ou não existe.")

    with conn:
        repo.encerrar_visita(conn, visita_id)
        repo.atualizar_situacao_cracha(
            conn,
            visita["cracha_id"],
            "disponivel",
        )


# ---------- setores ----------

def salvar_setor(
    conn,
    nome: str,
    lotacao_maxima: int,
    ativo: bool = True,
    setor_id: int | None = None,
) -> int:
    """Cria ou altera um setor, aplicando as regras de cadastro."""
    nome = normalizar_nome(nome)

    if not nome:
        raise RegraNegocioError("Informe o nome do setor.")

    if lotacao_maxima < 1:
        raise RegraNegocioError(
            "A lotação máxima precisa ser de pelo menos 1 visitante."
        )

    # Comparação em Python para tratar corretamente letras acentuadas.
    for setor in repo.listar_setores(conn):
        if (
            setor["nome"].casefold() == nome.casefold()
            and setor["id"] != setor_id
        ):
            raise RegraNegocioError(
                f"Já existe o setor {setor['nome']}."
            )

    with conn:
        if setor_id is None:
            return repo.inserir_setor(conn, nome, lotacao_maxima)

        presentes = repo.contar_presentes_no_setor(conn, setor_id)

        if lotacao_maxima < presentes:
            raise RegraNegocioError(
                f"Há {presentes} visitante(s) no setor agora. "
                "A lotação não pode ficar abaixo disso."
            )

        if not ativo and presentes:
            raise RegraNegocioError(
                "Não é possível desativar um setor com visitantes no prédio."
            )

        repo.atualizar_setor(
            conn,
            setor_id,
            nome,
            lotacao_maxima,
            ativo,
        )

        return setor_id


def excluir_setor(conn, setor_id: int) -> None:
    """Impede excluir setores que tenham histórico de visitas."""
    with conn:
        if repo.contar_visitas_do_setor(conn, setor_id):
            raise RegraNegocioError(
                "Este setor tem histórico de visitas e não pode ser excluído. "
                "Desative-o."
            )

        repo.excluir_setor(conn, setor_id)

    conn = sqlite3.connect(caminho)
    conn.row_factory = sqlite3.Row             # acessar colunas pelo nome

    conn.executescript(ESQUEMA.read_text(encoding="utf-8"))
    with conn:
        conn.executemany(
            "INSERT OR IGNORE INTO setor (nome, lotacao_maxima) VALUES (?, ?)",

        )
        conn.executemany(
            "INSERT OR IGNORE INTO cracha (codigo) VALUES (?)",
            [(f"V-{n:02d}",) for n in range(1, 11)],
        )

"""
Mapeamento ORM das tabelas da Portaria Digital, para comparação com o SQL explícito.

Não substitui repositorio.py: é um módulo isolado de estudo.
"""

from sqlalchemy import ForeignKey, create_engine, select
from sqlalchemy.orm import (
    DeclarativeBase,
    Mapped,
    Session,
    mapped_column,
    relationship,
    selectinload,
)


class Base(DeclarativeBase):
    pass


class Setor(Base):
    __tablename__ = "setor"

    id: Mapped[int] = mapped_column(primary_key=True)
    nome: Mapped[str]
    lotacao_maxima: Mapped[int]
    ativo: Mapped[int]


class Visitante(Base):
    __tablename__ = "visitante"

    id: Mapped[int] = mapped_column(primary_key=True)
    nome: Mapped[str]
    documento: Mapped[str]

    visitas: Mapped[list["Visita"]] = relationship(
        back_populates="visitante"
    )


class Cracha(Base):
    __tablename__ = "cracha"

    id: Mapped[int] = mapped_column(primary_key=True)
    codigo: Mapped[str]
    situacao: Mapped[str]


class Visita(Base):
    __tablename__ = "visita"

    id: Mapped[int] = mapped_column(primary_key=True)
    visitante_id: Mapped[int] = mapped_column(
        ForeignKey("visitante.id")
    )
    setor_id: Mapped[int] = mapped_column(
        ForeignKey("setor.id")
    )
    cracha_id: Mapped[int] = mapped_column(
        ForeignKey("cracha.id")
    )
    motivo: Mapped[str | None]
    entrada: Mapped[str]
    saida: Mapped[str | None]

    visitante: Mapped[Visitante] = relationship(
        back_populates="visitas"
    )
    setor: Mapped[Setor] = relationship()
    cracha: Mapped[Cracha] = relationship()


def criar_engine(caminho: str = "portaria.db", echo: bool = False):
    return create_engine(f"sqlite:///{caminho}", echo=echo)


def listar_presentes_orm(
    engine,
) -> list[tuple[int, str, str, str]]:
    with Session(engine) as sessao:
        consulta = (
            select(Visita)
            .where(Visita.saida.is_(None))
            .options(
                selectinload(Visita.visitante),
                selectinload(Visita.setor),
                selectinload(Visita.cracha),
            )
            .order_by(Visita.entrada)
        )

        return [
            (
                v.id,
                v.visitante.nome,
                v.setor.nome,
                v.cracha.codigo,
            )
            for v in sessao.scalars(consulta)
        ]

"""Persistência com SQL explícito e parametrizado.

Regra da casa: nenhum valor vindo da interface é concatenado em SQL.
Estas funções NÃO fazem commit; quem controla a transação é o serviço.
"""

import sqlite3

Linha = sqlite3.Row

# ---------- visitante ----------

def inserir_visitante(conn, nome: str, documento: str) -> int:
    cur = conn.execute(
        "INSERT INTO visitante (nome, documento) VALUES (?, ?)", (nome, documento)
    )
    return cur.lastrowid

def buscar_visitante_por_documento(conn, documento: str) -> Linha | None:
    return conn.execute(
        "SELECT id, nome, documento FROM visitante WHERE documento = ?", (documento,)
    ).fetchone()

def historico_do_visitante(conn, documento: str, limite: int = 5) -> list[Linha]:
    return conn.execute(
        """
        SELECT s.nome AS setor, v.entrada, v.saida
          FROM visita v
          JOIN visitante p ON p.id = v.visitante_id
          JOIN setor s     ON s.id = v.setor_id
         WHERE p.documento = ?
         ORDER BY v.entrada DESC
         LIMIT ?
        """,
        (documento, limite),
    ).fetchall()

# ---------- setor ----------

def listar_setores_ativos(conn) -> list[Linha]:
    return conn.execute(
        "SELECT id, nome, lotacao_maxima FROM setor WHERE ativo = 1 ORDER BY nome"
    ).fetchall()

def obter_setor(conn, setor_id: int) -> Linha | None:
    return conn.execute(
        "SELECT id, nome, lotacao_maxima, ativo FROM setor WHERE id = ?", (setor_id,)
    ).fetchone()

def contar_presentes_no_setor(conn, setor_id: int) -> int:
    return conn.execute(
        "SELECT COUNT(*) FROM visita WHERE setor_id = ? AND saida IS NULL", (setor_id,)
    ).fetchone()[0]

# ---------- crachá ----------

def buscar_cracha_disponivel(conn) -> Linha | None:
    return conn.execute(
        "SELECT id, codigo FROM cracha WHERE situacao = 'disponivel' "
        "ORDER BY codigo LIMIT 1"
    ).fetchone()

def atualizar_situacao_cracha(conn, cracha_id: int, situacao: str) -> None:
    conn.execute("UPDATE cracha SET situacao = ? WHERE id = ?", (situacao, cracha_id))

def contar_crachas_por_situacao(conn) -> dict[str, int]:
    linhas = conn.execute(
        "SELECT situacao, COUNT(*) AS total FROM cracha GROUP BY situacao"
    ).fetchall()
    return {l["situacao"]: l["total"] for l in linhas}

# ---------- visita ----------

def visita_aberta_do_visitante(conn, visitante_id: int) -> Linha | None:
    return conn.execute(
        "SELECT id FROM visita WHERE visitante_id = ? AND saida IS NULL",
        (visitante_id,),
    ).fetchone()

def inserir_visita(conn, visitante_id, setor_id, cracha_id, motivo, entrada) -> int:
    cur = conn.execute(
        "INSERT INTO visita (visitante_id, setor_id, cracha_id, motivo, entrada) "
        "VALUES (?, ?, ?, ?, ?)",
        (visitante_id, setor_id, cracha_id, motivo, entrada),
    )
    return cur.lastrowid

def obter_visita(conn, visita_id: int) -> Linha | None:
    return conn.execute(
        "SELECT id, visitante_id, cracha_id, entrada, saida FROM visita WHERE id = ?",
        (visita_id,),
    ).fetchone()

def encerrar_visita(conn, visita_id: int, saida: str) -> None:
    conn.execute("UPDATE visita SET saida = ? WHERE id = ?", (saida, visita_id))

def listar_presentes(conn) -> list[Linha]:
    """Quem está no prédio agora. É a lista usada numa evacuação."""
    return conn.execute(
        """
        SELECT v.id, p.nome, p.documento, s.nome AS setor, c.codigo AS cracha,
               v.entrada
          FROM visita v
          JOIN visitante p ON p.id = v.visitante_id
          JOIN setor s     ON s.id = v.setor_id
          JOIN cracha c    ON c.id = v.cracha_id
         WHERE v.saida IS NULL
         ORDER BY v.entrada
        """
    ).fetchall()

"""Regras de negócio da portaria.

A interface chama estas funções; elas validam, controlam a transação e
lançam RegraNegocioError com mensagens prontas para o usuário.
"""

from datetime import datetime

from portaria import repositorio as repo
from portaria.erros import RegraNegocioError
from portaria.validacao import cpf_valido, normalizar_documento, normalizar_nome

def agora() -> str:
    return datetime.now().isoformat(sep=" ", timespec="seconds")

def cadastrar_visitante(conn, nome: str, documento: str) -> int:
    nome = normalizar_nome(nome)
    doc = normalizar_documento(documento)
    if not nome:
        raise RegraNegocioError("Informe o nome do visitante.")
    if not cpf_valido(doc):
        raise RegraNegocioError("CPF inválido. Confira os 11 dígitos.")
    if repo.buscar_visitante_por_documento(conn, doc):
        raise RegraNegocioError("Já existe um visitante com este CPF.")
    with conn:  # commit no sucesso, rollback em exceção
        return repo.inserir_visitante(conn, nome, doc)

def registrar_entrada(conn, documento: str, setor_id: int, motivo: str = "",
                      momento: str | None = None) -> int:
    """Check-in: cria a visita E ocupa um crachá na MESMA transação (RN03)."""
    doc = normalizar_documento(documento)
    visitante = repo.buscar_visitante_por_documento(conn, doc)
    if visitante is None:
        raise RegraNegocioError("Visitante não cadastrado. Cadastre antes da entrada.")

    setor = repo.obter_setor(conn, setor_id)
    if setor is None or not setor["ativo"]:
        raise RegraNegocioError("Selecione um setor de destino válido.")

    # Aplicação desktop monousuário: as leituras abaixo e a escrita ficam no
    # mesmo bloco. Com vários terminais no mesmo banco, discutir BEGIN IMMEDIATE.
    with conn:
        if repo.visita_aberta_do_visitante(conn, visitante["id"]):
            raise RegraNegocioError(
                f"{visitante['nome']} já está no prédio. Registre a saída antes."
            )
        if repo.contar_presentes_no_setor(conn, setor_id) >= setor["lotacao_maxima"]:
            raise RegraNegocioError(
                f"{setor['nome']} atingiu a lotação de {setor['lotacao_maxima']} visitantes."
            )
        cracha = repo.buscar_cracha_disponivel(conn)
        if cracha is None:
            raise RegraNegocioError("Não há crachá disponível. Aguarde uma devolução.")

        visita_id = repo.inserir_visita(
            conn, visitante["id"], setor_id, cracha["id"],
            normalizar_nome(motivo) or None, momento or agora(),
        )
        repo.atualizar_situacao_cracha(conn, cracha["id"], "em_uso")
    return visita_id

def registrar_saida(conn, visita_id: int, momento: str | None = None) -> None:
    """Check-out: encerra a visita E devolve o crachá na mesma transação."""
    momento = momento or agora()
    with conn:
        visita = repo.obter_visita(conn, visita_id)
        if visita is None:
            raise RegraNegocioError("Visita não encontrada.")
        if visita["saida"] is not None:
            raise RegraNegocioError("Esta visita já foi encerrada.")
        if momento < visita["entrada"]:
            raise RegraNegocioError("A saída não pode ser anterior à entrada.")

        repo.encerrar_visita(conn, visita_id, momento)
        repo.atualizar_situacao_cracha(conn, visita["cracha_id"], "disponivel")

"""Funções puras de validação: não dependem da interface nem do banco."""

import re


def normalizar_nome(nome: str | None) -> str:
    """Remove espaços nas pontas e espaços repetidos no meio."""
    return " ".join((nome or "").split())


def normalizar_documento(valor: str | None) -> str:
    """Mantém apenas dígitos: '123.456.789-09' -> '12345678909'."""
    return re.sub(r"\D", "", valor or "")


def cpf_valido(valor: str | None) -> bool:
    """
    Valida os dígitos verificadores do CPF.

    Use apenas CPFs fictícios gerados para teste.
    Um CPF matematicamente válido pode pertencer a uma pessoa real.
    """
    d = normalizar_documento(valor)

    if len(d) != 11 or d == "0" * 11:
        return False

    for posicao in (9, 10):
        soma = sum(
            int(d[i]) * (posicao + 1 - i)
            for i in range(posicao)
        )
        digito = (soma * 10) % 11 % 10

        if digito != int(d[posicao]):
            return False

    return True


def mascarar_documento(valor: str | None) -> str:
    """Oculta parte do CPF para reduzir a exposição na tela."""
    d = normalizar_documento(valor)

    if len(d) != 11:
        return "***"

    return f"***.{d[3:6]}.{d[6:9]}-**"

def _luminancia(cor):
    cor = cor.lstrip("#")
    canais = [int(cor[i:i + 2], 16) / 255 for i in (0, 2, 4)]
    lin = [
        c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4
        for c in canais
    ]
    return 0.2126 * lin[0] + 0.7152 * lin[1] + 0.0722 * lin[2]


def contraste(cor_a, cor_b):
    """Razão de contraste WCAG entre duas cores hexadecimais."""
    la, lb = _luminancia(cor_a), _luminancia(cor_b)
    claro, escuro = max(la, lb), min(la, lb)
    return (claro + 0.05) / (escuro + 0.05)

    def _executar(self, funcao, *args):
        try:
            return funcao(*args)
        except RegraNegocioError as erro:
            self._avisar(str(erro))
        except sqlite3.Error:
            log.exception(
                "Falha de banco em %s",
                getattr(funcao, "__name__", "operação"),
            )
            QMessageBox.critical(
                self,
                "Portaria Digital",
                "Não foi possível acessar o banco de dados. "
                "Tente novamente ou avise o suporte.",
            )
        return FALHA

    def _avisar(self, texto):
        self.statusBar().showMessage(texto, 8000)
        QMessageBox.warning(self, "Portaria Digital", texto)

    def buscar_visitante(self):
        cpf = normalizar_documento(self.campo_cpf.text())
        visitante = self._executar(
            repo.buscar_visitante_por_documento, self.conn, cpf
        )
        if visitante is FALHA:
            return

        if visitante:
            self.campo_nome.setText(visitante["nome"])
            self.campo_nome.setReadOnly(True)
            self.mensagem.setText(
                "Visitante já cadastrado. Escolha o setor."
            )
        else:
            self.campo_nome.setReadOnly(False)

    def registrar_entrada(self):
        documento = self.campo_cpf.text()
        setor_id = self.combo_setor.currentData()

        if setor_id is None:
            self._avisar("Selecione o setor de destino.")
            return

        def operacao():
            cpf = normalizar_documento(documento)
            if repo.buscar_visitante_por_documento(self.conn, cpf) is None:
                servicos.cadastrar_visitante(
                    self.conn, self.campo_nome.text(), documento
                )
            return servicos.registrar_entrada(
                self.conn, documento, setor_id
            )

        if self._executar(operacao) is FALHA:
            self.mensagem.setText(
                "Entrada não registrada. Corrija os dados e tente de novo."
            )
            return

        self.limpar_campos()
        self.mensagem.setText(
            "Entrada registrada. Entregue o crachá indicado na tabela."
        )
        self.atualizar_tudo()

    def registrar_saida(self):
        linha = self.tabela.currentRow()
        if linha < 0:
            self._avisar("Selecione uma visita na tabela.")
            return

        visita_id = int(self.tabela.item(linha, 0).text())
        if self._executar(
            servicos.registrar_saida, self.conn, visita_id
        ) is not FALHA:
            self.mensagem.setText("Saída registrada. Crachá devolvido.")
            self.atualizar_tudo()

    def limpar_campos(self):
        self.campo_cpf.clear()
        self.campo_nome.clear()
        self.campo_nome.setReadOnly(False)
        self.combo_setor.setCurrentIndex(0)
        self.campo_cpf.setFocus()

    def carregar_setores(self):
        setores = self._executar(repo.listar_setores_ativos, self.conn)
        if setores is FALHA:
            return

        selecionado = self.combo_setor.currentData()
        self.combo_setor.clear()
        self.combo_setor.addItem("Selecione um setor", None)

        for setor in setores:
            ocupacao = self._executar(
                repo.contar_presentes_no_setor,
                self.conn,
                setor["id"],
            )
            if ocupacao is FALHA:
                return
            self.combo_setor.addItem(
                f"{setor['nome']} "
                f"({ocupacao}/{setor['lotacao_maxima']})",
                setor["id"],
            )

        self.combo_setor.setCurrentIndex(
            max(self.combo_setor.findData(selecionado), 0)
        )

    def atualizar_tudo(self):
        self.carregar_setores()
        self.atualizar_tabela()
        self.atualizar_historico()
        self.atualizar_setores()

    def atualizar_tabela(self):
        presentes = self._executar(repo.listar_presentes, self.conn)
        if presentes is FALHA:
            return

        termo = self.campo_filtro.text().strip().lower()
        preencher_tabela(
            self.tabela,
            [
                [
                    item["id"],
                    item["nome"],
                    mascarar_documento(item["documento"]),
                    item["setor"],
                    item["cracha"],
                    item["entrada"],
                ]
                for item in presentes
                if termo
                in f"{item['nome']} {item['setor']}".lower()
            ],
        )

        crachas = self._executar(
            repo.contar_crachas_por_situacao, self.conn
        )
        if crachas is FALHA:
            return

        self.contador.setText(
            f"No prédio: {len(presentes)} | "
            f"Crachás livres: {crachas.get('disponivel', 0)}"
        )

    def atualizar_historico(self):
        inicio = self.data_inicio.date().toString("yyyy-MM-dd")
        fim = self.data_fim.date().toString("yyyy-MM-dd")

        linhas = self._executar(
            repo.listar_historico,
            self.conn,
            inicio,
            fim,
            self.campo_busca.text(),
        )
        if linhas is FALHA:
            return

        preencher_tabela(
            self.tabela_historico,
            [
                [
                    item["id"],
                    item["nome"],
                    mascarar_documento(item["documento"]),
                    item["setor"],
                    item["cracha"],
                    item["entrada"],
                    item["saida"] or "no prédio",
                    item["minutos"],
                ]
                for item in linhas
            ],
        )

        resumo = self._executar(
            repo.resumo_por_setor, self.conn, inicio, fim
        )
        if resumo is FALHA:
            return

        partes = []
        for item in resumo:
            texto = f"{item['setor']}: {item['visitas']} visita(s)"
            if item["media_minutos"] is not None:
                texto += f", média de {item['media_minutos']} min"
            partes.append(texto)

        self.resumo.setText(
            "Resumo do período. " + "; ".join(partes)
        )

    def atualizar_setores(self):
        setores = self._executar(repo.listar_setores, self.conn)
        if setores is FALHA:
            return

        self._setores = setores
        preencher_tabela(
            self.tabela_setores,
            [
                [
                    setor["id"],
                    setor["nome"],
                    setor["lotacao_maxima"],
                    "Ativo" if setor["ativo"] else "Inativo",
                    setor["total_visitas"],
                ]
                for setor in setores
            ],
        )

    def setor_selecionado(self):
        linha = self.tabela_setores.currentRow()
        if linha < 0:
            self._avisar("Selecione um setor na tabela.")
            return None

        setor_id = int(self.tabela_setores.item(linha, 0).text())
        return next(
            (s for s in self._setores if s["id"] == setor_id),
            None,
        )

    def novo_setor(self):
        dialogo = DialogoSetor(self)
        if dialogo.exec() != QDialog.DialogCode.Accepted:
            return

        dados = dialogo.dados()
        resultado = self._executar(
            servicos.salvar_setor,
            self.conn,
            dados["nome"],
            dados["lotacao_maxima"],
        )
        if resultado is not FALHA:
            self.atualizar_tudo()

    def editar_setor(self):
        setor = self.setor_selecionado()
        if setor is None:
            return

        dialogo = DialogoSetor(self, setor)
        if dialogo.exec() != QDialog.DialogCode.Accepted:
            return

        dados = dialogo.dados()
        resultado = self._executar(
            servicos.salvar_setor,
            self.conn,
            dados["nome"],
            dados["lotacao_maxima"],
            dados["ativo"],
            setor["id"],
        )
        if resultado is not FALHA:
            self.atualizar_tudo()

    def excluir_setor(self):
        setor = self.setor_selecionado()
        if setor is None:
            return

        resposta = QMessageBox.question(
            self,
            "Portaria Digital",
            f"Excluir o setor {setor['nome']}?",
        )
        if resposta != QMessageBox.StandardButton.Yes:
            return

        resultado = self._executar(
            servicos.excluir_setor, self.conn, setor["id"]
        )
        if resultado is not FALHA:
            self.atualizar_tudo()
